function doGet(e) {
  const type = e.parameter.type;
  const action = e.parameter.action;
  if (type === 'report' || action === 'report') return getReadOnlyReport(e.parameter || {});
  if (type === 'salesDaily' || action === 'salesDaily') return getSalesDailyReport(e.parameter || {});
  if (type === 'authStatus') return authStatus(e.parameter.token || '');

  const user = sessionUser(e.parameter.token || '');
  if (!user) return json({ error: 'unauthorized' });
  const menu = String(e.parameter.menu || '');
  if (!canAccessMenu(user, menu) || !canReadData(type, menu)) return json({ error: 'forbidden' });

  if (type === 'leads') return getSheetData(SHEETS.leads);
  if (type === 'adSpend') return getSheetData(SHEETS.adSpend);
  if (type === 'firstRaw') return getSheetData(SHEETS.firstRaw);
  if (type === 'secondRaw') return getSheetData(SHEETS.secondRaw);
  if (type === 'projects') return getSheetData(SHEETS.projects);
  if (type === 'dashboardContracts') return getDashboardContractSummary();
  if (type === 'mapping') return getSheetData(SHEETS.mapping);
  if (type === 'kpiTargets') return getKpiTargetData();
  if (type === 'dataUpdatedAt') return getDataUpdatedAt();
  return json({ error: 'Invalid type' });
}

function getDataUpdatedAt() {
  const book = SpreadsheetApp.getActive();
  const updatedAt = DriveApp.getFileById(book.getId()).getLastUpdated();
  return json({ updatedAt: updatedAt.toISOString() });
}

function getDashboardContractSummary() {
  const byDate = {};
  uniqueSalesProjects(readRowsAsObjects(SHEETS.projects)).forEach(function(project) {
    if (!isSalesContract(project)) return;
    const date = salesProjectDate(project);
    if (!byDate[date]) byDate[date] = { date: date, contractCount: 0, contractAmount: 0 };
    byDate[date].contractCount += 1;
    byDate[date].contractAmount += salesProjectAmount(project);
  });
  return json(objectValues(byDate).sort(function(a, b) {
    return String(a.date).localeCompare(String(b.date));
  }));
}

function getReadOnlyReport(params) {
  const reportToken = String(params.reportToken || params.token || '');
  if (!isActiveReportToken(reportToken)) {
    return json({ error: 'forbidden' });
  }

  const from = normalizeReportDate(params.from || params.start);
  const to = normalizeReportDate(params.to || params.end);
  if (!from || !to) return json({ error: 'from/to required. Use YYYY-MM-DD.' });
  if (from > to) return json({ error: 'invalid range' });

  const leads = readRowsAsObjects(SHEETS.leads)
    .filter(function(row) { return isReportDateInRange(row.date, from, to); })
    .filter(function(row) { return !isExcludedReportLead(row); });
  const adSpendRows = readRowsAsObjects(SHEETS.adSpend)
    .filter(function(row) { return isReportDateInRange(row.date, from, to); });

  const fileUpdatedAt = formatReportDateTime(DriveApp.getFileById(SpreadsheetApp.getActive().getId()).getLastUpdated());
  const lastDbUploadedAt = latestReportTimestamp(leads, ['updatedAt', 'uploadedAt', 'dateOverrideAt', 'registeredAt', 'date']) || fileUpdatedAt;
  const lastAdSpendUpdatedAt = latestReportTimestamp(adSpendRows, ['updatedAt', 'uploadedAt', 'createdAt', 'date']) || fileUpdatedAt;
  const totals = blankReportMetric();
  const byDate = {};
  const byChannel = {};
  const bySubChannel = {};

  leads.forEach(function(row) {
    const date = String(row.date || '').slice(0, 10);
    const channel = reportChannel(row.channel || 'unknown', row);
    const subChannel = reportSubChannel(row.subChannel || row.source_raw || channel, channel);
    const stage = baseLeadStage(row.stage || row.dbTier || row.status);
    addLeadMetric(totals, stage);
    addLeadMetric(ensureReportMetric(byDate, date), stage);
    addLeadMetric(ensureReportMetric(byChannel, channel), stage);
    addLeadMetric(ensureReportMetric(bySubChannel, channel + '|' + subChannel, {
      channel: channel,
      subChannel: subChannel
    }), stage);
  });

  adSpendRows.forEach(function(row) {
    const date = String(row.date || '').slice(0, 10);
    const channel = reportChannel(row.channel || 'unknown', row);
    const subChannel = reportSubChannel(row.subChannel || row.source_raw || channel, channel);
    const amount = reportNumber(row.amount || row.cost || row['광고비'] || row['비용']);
    addSpendMetric(totals, amount);
    addSpendMetric(ensureReportMetric(byDate, date), amount);
    addSpendMetric(ensureReportMetric(byChannel, channel), amount);
    addSpendMetric(ensureReportMetric(bySubChannel, channel + '|' + subChannel, {
      channel: channel,
      subChannel: subChannel
    }), amount);
  });

  finalizeReportMetric(totals);
  Object.keys(byDate).forEach(function(key) { finalizeReportMetric(byDate[key]); });
  Object.keys(byChannel).forEach(function(key) { finalizeReportMetric(byChannel[key]); });
  Object.keys(bySubChannel).forEach(function(key) { finalizeReportMetric(bySubChannel[key]); });

  return json({
    success: true,
    timezone: REPORT_TIMEZONE,
    range: { from: from, to: to },
    updatedAt: fileUpdatedAt,
    lastDbUploadedAt: lastDbUploadedAt,
    lastAdSpendUpdatedAt: lastAdSpendUpdatedAt,
    totals: totals,
    byDate: objectValues(byDate).sort(function(a, b) { return String(a.date).localeCompare(String(b.date)); }),
    byChannel: objectValues(byChannel).sort(sortReportMetric),
    bySubChannel: objectValues(bySubChannel).sort(sortReportMetric),
    criteria: {
      adSpendVat: 'AD_SPEND 시트에 저장된 금액 그대로 사용합니다. 현재 대시보드 기준은 입력 광고비 금액 기준이며, VAT 포함/별도 여부는 업로드 금액 기준을 따릅니다.',
      duplicatePolicy: 'DASHBOARD_LEADS 원장 기준으로 이미 정제·중복 제거된 행만 집계합니다. 상태가 invalid/test/duplicate/deleted인 행은 제외합니다.',
      validDbPolicy: '유효 DB는 1차 DB와 2차 DB 합계입니다. 리타겟 DB는 총 DB에는 포함하지만 CPL 산정용 유효 DB에서는 제외합니다.',
      cplFormula: 'CPL = 총 광고비 / 유효 DB. 유효 DB가 0이면 CPL은 0으로 반환합니다.',
      datePolicy: 'date 컬럼 기준으로 from/to 기간을 포함 조회합니다. 모든 날짜·시각 표기는 Asia/Seoul 기준입니다.',
      privacyPolicy: '개인정보, 연락처, 이름, 주소, 원본 DB 행은 반환하지 않습니다.'
    },
    tokenPolicy: {

function getSalesDailyReport(params) {
  const reportToken = String(params.reportToken || params.token || '');
  if (!isActiveReportToken(reportToken)) return json({ error: 'forbidden' });

  const from = normalizeReportDate(params.from || params.start);
  const to = normalizeReportDate(params.to || params.end);
  if (!from || !to) return json({ error: 'from/to required. Use YYYY-MM-DD.' });
  if (from > to) return json({ error: 'invalid range' });

  const allLeads = readRowsAsObjects(SHEETS.leads);
  const allProjects = readRowsAsObjects(SHEETS.projects);
  const assignmentRows = readRowsAsObjects(SHEETS.secondRaw);
  allLeads.forEach(function(row, index) { row.__salesIndex = index; });
  const assignmentLookup = buildSalesAssignmentLookup(allLeads.concat(assignmentRows));
  const periodLeads = allLeads.filter(function(row) {
    return isReportDateInRange(row.date, from, to);
  });
  const periodProjects = uniqueSalesProjects(allProjects).filter(function(row) {
    return isReportDateInRange(salesProjectDate(row), from, to) && isSalesContract(row);
  });

  const estimateAliases = [
    '견적서 발송', '견적서발송', '견적 발송', '견적발송', '견적서 발송일', '견적발송일',
    'quoteSent', 'quoteSentAt', 'estimateSent', 'estimateSentAt'
  ];
  const measurementAliases = [
    '실측 완료', '실측완료', '실측 완료일', '실측완료일', '실측일',
    'measurementCompleted', 'measurementCompletedAt', 'surveyCompleted', 'surveyCompletedAt'
  ];
  const estimateProjectField = findAvailableField([allProjects], estimateAliases);
  const estimateLeadField = findAvailableField([allLeads], estimateAliases);
  const estimateField = estimateProjectField || estimateLeadField;
  const estimateSource = estimateProjectField ? 'PROJECTS' : (estimateLeadField ? 'DASHBOARD_LEADS' : null);
  const measurementProjectField = findAvailableField([allProjects], measurementAliases);
  const measurementLeadField = findAvailableField([allLeads], measurementAliases);
  const measurementField = measurementProjectField || measurementLeadField;
  const measurementSource = measurementProjectField ? 'PROJECTS' : (measurementLeadField ? 'DASHBOARD_LEADS' : null);

  const buckets = {};
  const activeLeads = [];
  const explicitExclusions = { invalid: 0, test: 0, duplicate: 0, deleted: 0 };

  periodLeads.forEach(function(row, index) {
    const status = salesLeadStatus(row);
    const rawOwner = resolveSalesOwner(row, assignmentLookup);
    const rawAssignmentCategory = classifySalesAssignment(row, assignmentLookup, rawOwner);
    const rawBucket = ensureSalesBucket(buckets, salesLeadDimensions(row, rawOwner, rawAssignmentCategory));
    rawBucket.rawDb += 1;

    if (Object.prototype.hasOwnProperty.call(explicitExclusions, status)) {
      explicitExclusions[status] += 1;
      addSalesExclusion(rawBucket, status, 1);
      return;
    }

    activeLeads.push(row);
  });

  const recognizedByContact = {};
  allLeads.forEach(function(row) {
    const status = salesLeadStatus(row);
    if (status === 'invalid' || status === 'test' || status === 'duplicate' || status === 'deleted') return;
    const key = salesLeadContactKey(row);
    const current = recognizedByContact[key];
    if (!current || compareSalesLead(row, current) > 0) recognizedByContact[key] = row;
  });

  const recognizedLeads = objectValues(recognizedByContact).filter(function(row) {
    return isReportDateInRange(row.date, from, to);
  });
  const deduplicatedCount = Math.max(0, activeLeads.length - recognizedLeads.length);
  const duplicatesByBucket = {};
  activeLeads.forEach(function(row) {
    const winner = recognizedByContact[salesLeadContactKey(row)];
    if (winner === row) return;
    const duplicateOwner = resolveSalesOwner(row, assignmentLookup);
    const duplicateCategory = classifySalesAssignment(row, assignmentLookup, duplicateOwner);
    const dimensions = salesLeadDimensions(row, duplicateOwner, duplicateCategory);
    const bucketKey = salesBucketKey(dimensions);
    duplicatesByBucket[bucketKey] = (duplicatesByBucket[bucketKey] || 0) + 1;
  });
  Object.keys(duplicatesByBucket).forEach(function(key) {
    const dimensions = salesBucketDimensionsFromKey(key);
    addSalesExclusion(ensureSalesBucket(buckets, dimensions), 'duplicate', duplicatesByBucket[key]);
  });

  recognizedLeads.forEach(function(row) {
    const owner = resolveSalesOwner(row, assignmentLookup);
    const assignmentCategory = classifySalesAssignment(row, assignmentLookup, owner);
    const bucket = ensureSalesBucket(buckets, salesLeadDimensions(row, owner, assignmentCategory));
    bucket.recognizedDb += 1;
    if (assignmentCategory === 'assigned') bucket.assignedDb += 1;
    else if (assignmentCategory === 'explicit_assignment_canceled') {
      bucket.explicitAssignmentCanceledDb += 1;
      bucket.assignmentCanceledDb += 1;
      bucket.assignmentCanceledOrExcludedDb += 1;
    }
    else if (assignmentCategory === 'blank_owner_excluded') {
      bucket.blankOwnerExcludedDb += 1;
      bucket.assignmentCanceledOrExcludedDb += 1;
    }
    else if (assignmentCategory === 'pending_assignment') bucket.pendingAssignmentDb += 1;
    else if (assignmentCategory === 'mapping_failed') bucket.ownerMappingFailedDb += 1;
    else if (assignmentCategory === 'truly_unassigned') bucket.trulyUnassignedDb += 1;
    else bucket.otherUnassignedDb += 1;
    if (estimateSource === 'DASHBOARD_LEADS') bucket.quoteSentCount += isCompletedField(row[estimateField]) ? 1 : 0;
    if (measurementSource === 'DASHBOARD_LEADS') bucket.measurementCompletedCount += isCompletedField(row[measurementField]) ? 1 : 0;
  });

  const leadLookup = buildSalesLeadLookup(allLeads);
  periodProjects.forEach(function(project) {
    const matchedLead = matchSalesProjectLead(project, leadLookup);
    const dimensions = salesProjectDimensions(project, matchedLead, assignmentLookup);
    const bucket = ensureSalesBucket(buckets, dimensions);
    bucket.contractCount += 1;
    bucket.contractAmount += salesProjectAmount(project);
    if (estimateSource === 'PROJECTS') bucket.quoteSentCount += isCompletedField(project[estimateField]) ? 1 : 0;
    if (measurementSource === 'PROJECTS') bucket.measurementCompletedCount += isCompletedField(project[measurementField]) ? 1 : 0;
  });

  const rows = objectValues(buckets)
    .map(function(bucket) {
      if (!estimateField) bucket.quoteSentCount = 'unavailable';
      if (!measurementField) bucket.measurementCompletedCount = 'unavailable';
      return bucket;
    })
    .sort(function(a, b) {
      return String(a.date).localeCompare(String(b.date)) ||
        String(a.salesManager).localeCompare(String(b.salesManager)) ||
        String(a.inflowType).localeCompare(String(b.inflowType)) ||
        String(a.channel).localeCompare(String(b.channel));
    });

  const totals = rows.reduce(function(total, row) {
    total.rawDb += row.rawDb;
    total.recognizedDb += row.recognizedDb;
    total.assignedDb += row.assignedDb;
    total.explicitAssignmentCanceledDb += row.explicitAssignmentCanceledDb;
    total.blankOwnerExcludedDb += row.blankOwnerExcludedDb;
    total.assignmentCanceledOrExcludedDb += row.assignmentCanceledOrExcludedDb;
    total.pendingAssignmentDb += row.pendingAssignmentDb;
    total.assignmentCanceledDb += row.assignmentCanceledDb;
    total.trulyUnassignedDb += row.trulyUnassignedDb;
    total.ownerMappingFailedDb += row.ownerMappingFailedDb;
    total.otherUnassignedDb += row.otherUnassignedDb;
    total.contractCount += row.contractCount;
    total.contractAmount += row.contractAmount;
    total.excluded.duplicate += row.excluded.duplicate;
    total.excluded.test += row.excluded.test;
    total.excluded.invalid += row.excluded.invalid;
    total.excluded.deleted += row.excluded.deleted;
    if (typeof row.quoteSentCount === 'number') total.quoteSentCount += row.quoteSentCount;
    if (typeof row.measurementCompletedCount === 'number') total.measurementCompletedCount += row.measurementCompletedCount;
    return total;
  }, {
    rawDb: 0,
    recognizedDb: 0,
    assignedDb: 0,
    explicitAssignmentCanceledDb: 0,
    blankOwnerExcludedDb: 0,
    assignmentCanceledOrExcludedDb: 0,
    pendingAssignmentDb: 0,
    assignmentCanceledDb: 0,
    trulyUnassignedDb: 0,
    ownerMappingFailedDb: 0,
    otherUnassignedDb: 0,
    contractCount: 0,
    contractAmount: 0,
    excluded: { duplicate: 0, test: 0, invalid: 0, deleted: 0 },
    quoteSentCount: 0,
    measurementCompletedCount: 0
  });
  if (!estimateField) totals.quoteSentCount = 'unavailable';
  if (!measurementField) totals.measurementCompletedCount = 'unavailable';
  const assignmentClassifiedDb = totals.assignedDb + totals.explicitAssignmentCanceledDb +
    totals.blankOwnerExcludedDb + totals.pendingAssignmentDb + totals.trulyUnassignedDb +
    totals.ownerMappingFailedDb + totals.otherUnassignedDb;

  const salesManagerMap = {};
  rows.forEach(function(row) {
    const manager = row.salesManager || '미배정';
    if (!salesManagerMap[manager]) {
      salesManagerMap[manager] = {
        salesManager: manager,
        recognizedDb: 0,
        assignedDb: 0,
        explicitAssignmentCanceledDb: 0,
        blankOwnerExcludedDb: 0,
        assignmentCanceledOrExcludedDb: 0,
        pendingAssignmentDb: 0,
        assignmentCanceledDb: 0,
        trulyUnassignedDb: 0,
        ownerMappingFailedDb: 0,
        otherUnassignedDb: 0,
        contractCount: 0,
        contractAmount: 0
      };
    }
    salesManagerMap[manager].recognizedDb += row.recognizedDb;
    salesManagerMap[manager].assignedDb += row.assignedDb;
    salesManagerMap[manager].explicitAssignmentCanceledDb += row.explicitAssignmentCanceledDb;
    salesManagerMap[manager].blankOwnerExcludedDb += row.blankOwnerExcludedDb;
    salesManagerMap[manager].assignmentCanceledOrExcludedDb += row.assignmentCanceledOrExcludedDb;
    salesManagerMap[manager].pendingAssignmentDb += row.pendingAssignmentDb;
    salesManagerMap[manager].assignmentCanceledDb += row.assignmentCanceledDb;
    salesManagerMap[manager].trulyUnassignedDb += row.trulyUnassignedDb;
    salesManagerMap[manager].ownerMappingFailedDb += row.ownerMappingFailedDb;
    salesManagerMap[manager].otherUnassignedDb += row.otherUnassignedDb;
    salesManagerMap[manager].contractCount += row.contractCount;
    salesManagerMap[manager].contractAmount += row.contractAmount;
  });
  const bySalesManager = objectValues(salesManagerMap).sort(function(a, b) {
    return b.contractCount - a.contractCount || b.assignedDb - a.assignedDb || String(a.salesManager).localeCompare(String(b.salesManager));
  });
  const unassignedReasonBreakdown = {
    explicitAssignmentCanceled: totals.explicitAssignmentCanceledDb,
    blankOwnerExcluded: totals.blankOwnerExcludedDb,
    assignmentCanceledOrExcluded: totals.assignmentCanceledOrExcludedDb,
    pendingAssignment: totals.pendingAssignmentDb,
    trulyUnassigned: totals.trulyUnassignedDb,
    mappingFailed: totals.ownerMappingFailedDb,
    other: totals.otherUnassignedDb
  };
  const assignmentBreakdownMap = {};
  rows.forEach(function(row) {
    const key = [row.date, row.inflowType, row.channel].join('|');
    if (!assignmentBreakdownMap[key]) {
      assignmentBreakdownMap[key] = {
        date: row.date,
        inflowType: row.inflowType,
        channel: row.channel,
        channelLabel: row.channelLabel,
        recognizedDb: 0,
        assignedDb: 0,
        explicitAssignmentCanceledDb: 0,
        blankOwnerExcludedDb: 0,
        assignmentCanceledOrExcludedDb: 0,
        pendingAssignmentDb: 0,
        assignmentCanceledDb: 0,
        trulyUnassignedDb: 0,
        ownerMappingFailedDb: 0,
        otherUnassignedDb: 0
      };
    }
    const item = assignmentBreakdownMap[key];
    item.recognizedDb += row.recognizedDb;
    item.assignedDb += row.assignedDb;
    item.explicitAssignmentCanceledDb += row.explicitAssignmentCanceledDb;
    item.blankOwnerExcludedDb += row.blankOwnerExcludedDb;
    item.assignmentCanceledOrExcludedDb += row.assignmentCanceledOrExcludedDb;
    item.pendingAssignmentDb += row.pendingAssignmentDb;
    item.assignmentCanceledDb += row.assignmentCanceledDb;
    item.trulyUnassignedDb += row.trulyUnassignedDb;
    item.ownerMappingFailedDb += row.ownerMappingFailedDb;
    item.otherUnassignedDb += row.otherUnassignedDb;
  });
  const assignmentBreakdownByDateChannel = objectValues(assignmentBreakdownMap).sort(function(a, b) {
    return String(a.date).localeCompare(String(b.date)) || String(a.channel).localeCompare(String(b.channel));
  });

  const fileUpdatedAt = formatReportDateTime(DriveApp.getFileById(SpreadsheetApp.getActive().getId()).getLastUpdated());
  const lastUpdatedAt = latestReportTimestamp(
    periodLeads.concat(periodProjects),
    ['updatedAt', 'uploadedAt', 'dateOverrideAt', 'registeredAt', 'contractDate', 'date']
  ) || fileUpdatedAt;

  return json({
    success: true,
    report: 'sales-daily',
    timezone: REPORT_TIMEZONE,
    range: { from: from, to: to, inclusive: true },
    lastUpdatedAt: lastUpdatedAt,
    totals: totals,
    availability: {
      quoteSentCount: estimateField ? { status: 'available', source: estimateSource, field: estimateField } : 'unavailable',
      measurementCompletedCount: measurementField ? { status: 'available', source: measurementSource, field: measurementField } : 'unavailable'
    },
    bySalesManager: bySalesManager,
    assignmentSummary: {
      assignedDb: totals.assignedDb,
      explicitAssignmentCanceledDb: totals.explicitAssignmentCanceledDb,
      blankOwnerExcludedDb: totals.blankOwnerExcludedDb,
      assignmentCanceledOrExcludedDb: totals.assignmentCanceledOrExcludedDb,
      pendingAssignmentDb: totals.pendingAssignmentDb,
      assignmentCanceledDb: totals.assignmentCanceledDb,
      trulyUnassignedDb: totals.trulyUnassignedDb,
      ownerMappingFailedDb: totals.ownerMappingFailedDb,
      otherUnassignedDb: totals.otherUnassignedDb
    },
    unassignedReasonBreakdown: unassignedReasonBreakdown,
    assignmentBreakdownByDateChannel: assignmentBreakdownByDateChannel,
    rows: rows,
    reconciliation: {
      recognizedDb: recognizedLeads.length,
      sourceRows: periodLeads.length,
      activeRowsBeforeContactDedupe: activeLeads.length,
      contactDedupeExcluded: deduplicatedCount,
      explicitStatusExcluded: explicitExclusions,
      assignmentClassifiedDb: assignmentClassifiedDb,
      assignmentClassificationMatchesRecognized: assignmentClassifiedDb === totals.recognizedDb,
      dashboardComparisonRule: '선택 기간 내 활성 리드를 연락처 기준으로 묶고 최종 단계 1건만 인정합니다.'
    },
    criteria: {
      source: 'DASHBOARD_LEADS 및 PROJECTS 시트',
      datePolicy: 'Asia/Seoul 기준 DB는 date, 계약은 계약금입금일(depositDate)을 사용하며 from/to 양끝 날짜를 포함합니다.',
      recognizedDbPolicy: 'invalid, test, duplicate, deleted 상태를 제외한 뒤 연락처 기준으로 중복 제거하고 최종 단계 1건을 인정합니다.',
      assignedDbPolicy: '인정 DB를 컨설팅번호 우선, 연락처 보조로 컨설팅리스트 담당자와 연결하고 유효한 영업지점장이 있는 건입니다.',
      assignmentClassificationPolicy: '영업담당자가 있으면 assigned, 명시적 배정취소는 explicit_assignment_canceled, 담당자 공란은 당일 유입일 때 pending_assignment·익일 이후 blank_owner_excluded로 분류합니다. 원본 담당자 값이 있지만 매핑이 실패하면 mapping_failed, 명시적 미배정 상태가 확인된 경우에만 truly_unassigned로 분류합니다.',

function hasSalesOwner(value) {
  const owner = String(value || '').trim();
  if (!owner || owner === '-') return false;
  const normalized = owner.toLowerCase().replace(/\s/g, '');
  if (['미배정', '시스템', 'system', '업로드', '없음', '미지정', 'null', 'undefined'].indexOf(normalized) >= 0) return false;
  return !/배정(취소|해제|회수|철회)/.test(normalized);
}

function cleanSalesOwner(value) {
  return hasSalesOwner(value) ? String(value || '').trim() : '미배정';
}

  }
  return '';
}

function isSalesAssignmentCanceled(row) {
  if (/배정\s*(취소|해제|회수|철회)/.test(salesAssignmentStatusText(row))) return true;
  return salesAssignmentLifecycleStatuses(row).some(function(value) {

    return normalized === '취소' || normalized === '배정취소' || normalized === '배정해제' ||
      normalized === '배정회수' || normalized === '배정철회';
  });
}

function isSalesExplicitlyUnassigned(row) {
  return salesAssignmentLifecycleStatuses(row).some(function(value) {

function setSalesAssignment(map, key, owner, sortValue) {
  if (!key) return;
  const current = map[key];
  if (!current || String(sortValue || '').localeCompare(String(current.sortValue || '')) >= 0) {
    map[key] = { owner: owner, sortValue: sortValue || '' };
  }
}

function setSalesAssignmentEvidence(map, key, row) {
  if (!key) return;
  const current = map[key] || {
    assignmentCanceled: false,
    rawOwnerPresent: false,
    validOwnerPresent: false,
    explicitlyUnassigned: false
  };
  const rawOwner = salesOwnerValue(row);
  if (String(rawOwner || '').trim()) current.rawOwnerPresent = true;
  if (hasSalesOwner(rawOwner)) current.validOwnerPresent = true;
  if (isSalesAssignmentCanceled(row)) current.assignmentCanceled = true;
  if (isSalesExplicitlyUnassigned(row)) current.explicitlyUnassigned = true;
  map[key] = current;
}

function buildSalesAssignmentLookup(rows) {
  const byConsulting = {};
  const byPhone = {};
  const evidenceByConsulting = {};
  const evidenceByPhone = {};
  (rows || []).forEach(function(row) {
    const owner = salesOwnerValue(row);
    const consultingNumber = salesConsultingNumberValue(row);
    const phone = normalizePhone(salesPhoneValue(row));
    setSalesAssignmentEvidence(evidenceByConsulting, consultingNumber, row);
    setSalesAssignmentEvidence(evidenceByPhone, phone, row);
    if (!hasSalesOwner(owner)) return;
    const sortValue = salesAssignmentSortValue(row);
    setSalesAssignment(byConsulting, consultingNumber, owner, sortValue);
    setSalesAssignment(byPhone, phone, owner, sortValue);
  });
  return {
    byConsulting: byConsulting,
    byPhone: byPhone,
    evidenceByConsulting: evidenceByConsulting,
    evidenceByPhone: evidenceByPhone
  };
}

function resolveSalesOwner(row, lookup) {
  const ownOwner = salesOwnerValue(row);
  if (hasSalesOwner(ownOwner)) return ownOwner;
  if (!lookup) return '';
  const consultingNumber = salesConsultingNumberValue(row);
  const phone = normalizePhone(salesPhoneValue(row));
  const matched = (consultingNumber && lookup.byConsulting[consultingNumber]) || (phone && lookup.byPhone[phone]);
  return matched ? matched.owner : '';
}

function mergeSalesAssignmentEvidence(target, source) {
  if (!source) return target;
  target.assignmentCanceled = target.assignmentCanceled || source.assignmentCanceled;
  target.rawOwnerPresent = target.rawOwnerPresent || source.rawOwnerPresent;
  target.validOwnerPresent = target.validOwnerPresent || source.validOwnerPresent;
  target.explicitlyUnassigned = target.explicitlyUnassigned || source.explicitlyUnassigned;
  return target;
}

function resolveSalesAssignmentEvidence(row, lookup) {
  const ownOwner = salesOwnerValue(row);
  const evidence = {
    assignmentCanceled: isSalesAssignmentCanceled(row),
    rawOwnerPresent: Boolean(String(ownOwner || '').trim()),
    validOwnerPresent: hasSalesOwner(ownOwner),
    explicitlyUnassigned: isSalesExplicitlyUnassigned(row)
  };
  if (!lookup) return evidence;
  const consultingNumber = salesConsultingNumberValue(row);
  const phone = normalizePhone(salesPhoneValue(row));
  if (consultingNumber) mergeSalesAssignmentEvidence(evidence, lookup.evidenceByConsulting[consultingNumber]);
  if (phone) mergeSalesAssignmentEvidence(evidence, lookup.evidenceByPhone[phone]);
  return evidence;
}

function classifySalesAssignment(row, lookup, resolvedOwner) {
  if (hasSalesOwner(resolvedOwner)) return 'assigned';
  const evidence = resolveSalesAssignmentEvidence(row, lookup);
  if (evidence.assignmentCanceled) return 'explicit_assignment_canceled';
  if (evidence.explicitlyUnassigned) return 'truly_unassigned';
  if (evidence.rawOwnerPresent) return 'mapping_failed';
  if (normalizeReportDate(row && row.date) === reportTodayDate()) return 'pending_assignment';
  return 'blank_owner_excluded';
}

}

function betterSalesLookupLead(current, next) {
  if (!current) return next;
  const currentOwner = hasSalesOwner(salesOwnerValue(current));
  const nextOwner = hasSalesOwner(salesOwnerValue(next));
  if (currentOwner !== nextOwner) return nextOwner ? next : current;
  return compareSalesLead(next, current) > 0 ? next : current;
}

    if (phone) byPhone[phone] = betterSalesLookupLead(byPhone[phone], row);
  });
  return { byConsulting: byConsulting, byPhone: byPhone };
}

function findAvailableField(rowGroups, aliases) {
  const aliasKeys = aliases.map(normalizeSalesHeader);
  for (let groupIndex = 0; groupIndex < rowGroups.length; groupIndex++) {
    const rows = rowGroups[groupIndex] || [];
    for (let rowIndex = 0; rowIndex < rows.length; rowIndex++) {
      const keys = Object.keys(rows[rowIndex] || {});
      for (let keyIndex = 0; keyIndex < keys.length; keyIndex++) {
        if (aliasKeys.indexOf(normalizeSalesHeader(keys[keyIndex])) >= 0) return keys[keyIndex];
      }
    }
  }
  return '';
}

function readRowsAsObjects(sheetName) {
  const sheet = SpreadsheetApp.getActive().getSheetByName(sheetName);
  if (!sheet) return [];
  const values = sheet.getDataRange().getDisplayValues();
  if (!values || values.length <= 1) return [];
  const headers = values.shift();
  return values
    .filter(function(row) { return row.some(function(cell) { return cell !== ''; }); })
    .map(function(row) {
      const obj = {};
      headers.forEach(function(header, index) { obj[header] = row[index]; });
      return obj;
    });
}

function normalizeReportDate(value) {
  const text = String(value || '').trim();
  const match = text.match(/^(\d{4})[-/.](\d{1,2})[-/.](\d{1,2})/);
  if (!match) return '';
  return match[1] + '-' + ('0' + match[2]).slice(-2) + '-' + ('0' + match[3]).slice(-2);
}

function normalizeReportDateTime(value) {
  const text = String(value || '').trim();
  if (!text) return '';
  const date = normalizeReportDate(text);
  const timeMatch = text.replace('T', ' ').match(/\d{4}[-/.]\d{1,2}[-/.]\d{1,2}\s+(\d{1,2}):(\d{1,2})(?::(\d{1,2}))?/);
  if (!date || !timeMatch) return date;
  return date + ' ' + ('0' + timeMatch[1]).slice(-2) + ':' + ('0' + timeMatch[2]).slice(-2) + ':' + ('0' + (timeMatch[3] || '00')).slice(-2);
}

function formatReportDateTime(value) {
  try {
    return Utilities.formatDate(new Date(value), REPORT_TIMEZONE, 'yyyy-MM-dd HH:mm:ss');
  } catch (err) {
    return normalizeReportDateTime(value);
  }
}

function latestReportTimestamp(rows, keys) {
  let latestRaw = '';
  let latestKey = '';
  rows.forEach(function(row) {
    keys.forEach(function(key) {
      const raw = String(row[key] || '').trim();
      const normalized = normalizeReportDateTime(raw);
      if (normalized && normalized > latestKey) {
        latestKey = normalized;
        latestRaw = raw;
      }
    });
  });
  return latestRaw ? formatReportDateTime(latestRaw) : '';
}

function isReportDateInRange(value, from, to) {
  const date = normalizeReportDate(value);
  return date && date >= from && date <= to;
}

    row && row.source_raw,
    row && row.sourceRaw,
    row && row.utm_source,
    row && row.utmSource,
    row && row.originPath
  ].map(function(item) { return String(item || '').trim().toLowerCase(); });
  const keys = rawValues.map(function(item) { return item.replace(/[\s_\-\/()\[\].]/g, ''); });
  if (keys.indexOf('gptad') >= 0) return 'chatgpt';
  if (keys.indexOf('ezpz') >= 0) return 'ezpz';
  if (keys.indexOf('direct') >= 0 || keys.indexOf('다이렉트') >= 0) return 'direct';
  return String(value || 'unknown').trim() || 'unknown';
}

function reportSubChannel(value, channel) {
  const label = String(value || 'unknown').trim() || 'unknown';
  const normalized = label.toLowerCase().replace(/[\s_\-\/()\[\].]/g, '');
  if (channel === 'chatgpt' && (normalized === 'gptad' || normalized === '기타' || normalized === 'etc')) return 'Chat-GPT';
  if (channel === 'ezpz') return 'EZPZ';
  if (channel === 'direct' && (normalized === 'direct' || normalized === '다이렉트' || normalized === '기타' || normalized === 'etc')) return '홈페이지 직접유입';
  if (normalized.indexOf('네이버') >= 0 || normalized.indexOf('naver') >= 0) {
    if (normalized.indexOf('gfa') >= 0) return '네이버 GFA';
    if (normalized.indexOf('브랜드검색') >= 0 || normalized.indexOf('brand') >= 0) return '네이버 브랜드검색';
    if (normalized.indexOf('sa') >= 0 || normalized.indexOf('파워링크') >= 0) return '네이버 SA';
  }
  return label;
}

function blankReportMetric(extra) {
  const base = {
    adSpend: 0,
    db: 0,
    retarget: 0,
    firstDb: 0,
    secondDb: 0,
    validDb: 0,
    cpl: 0
  };
  if (extra) {
    Object.keys(extra).forEach(function(key) { base[key] = extra[key]; });
  }
  return base;
}

function ensureReportMetric(bucket, key, extra) {
  if (!bucket[key]) {
    bucket[key] = blankReportMetric(extra || {});
    if (key && !bucket[key].date && String(key).indexOf('|') < 0) {
      if (/^\d{4}-\d{2}-\d{2}$/.test(String(key))) bucket[key].date = key;
      else bucket[key].channel = key;
    }
  }
  return bucket[key];
}

function addLeadMetric(metric, stage) {
  metric.db += 1;
  if (stage === 'retarget') metric.retarget += 1;
  else if (stage === 'first') {
    metric.firstDb += 1;
    metric.validDb += 1;
  } else if (stage === 'second') {
    metric.secondDb += 1;
    metric.validDb += 1;
  } else {
    metric.validDb += 1;
  }
}

  metric.adSpend += amount;
}

function finalizeReportMetric(metric) {
  metric.cpl = metric.validDb > 0 ? Math.round(metric.adSpend / metric.validDb) : 0;
}

}

function sortReportMetric(a, b) {
  if (b.validDb !== a.validDb) return b.validDb - a.validDb;

}

function doPost(e) {
  try {
    const body = JSON.parse(e.postData.contents);
    const type = body.type;
    const rows = body.rows || [];

    if (type === 'authSetup') return authSetup(body);
    if (type === 'authLogin') return authLogin(body);
    if (type === 'authLogout') return authLogout(body.token || '');

    const user = sessionUser(body.token || '');
    if (!user) return json({ error: 'unauthorized' });
    if (type === 'authListUsers') return authListUsers(user);
    if (type === 'authCreateUser') return authCreateUser(user, body);
    if (type === 'authUpdateUser') return authUpdateUser(user, body);

    const menu = resolveWriteMenu(type, String(body.menu || ''));
    if (!canAccessMenu(user, menu) || !canWriteData(type, menu)) return json({ error: 'forbidden' });

    if (type === 'leadUploadContext') return getLeadUploadContext(body);
    if (type === 'leads') return appendRowsUnique(SHEETS.leads, rows, 'leads');
    if (type === 'adSpend') return appendRowsUnique(SHEETS.adSpend, rows, 'adSpend');
    if (type === 'adSpendReplace') return replaceAdSpendRows(rows);
    if (type === 'updateAdSpend') return updateAdSpendRow(body);
    if (type === 'deleteAdSpend') return deleteAdSpendRow(body);
    if (type === 'firstRaw') return upsertRowsUnique(SHEETS.firstRaw, rows, 'firstRaw');
    if (type === 'secondRaw') return upsertRowsUnique(SHEETS.secondRaw, rows, 'secondRaw');
    if (type === 'projects') return upsertRowsUnique(SHEETS.projects, rows, 'projects');
    if (type === 'projectsReplace') return replaceProjectRows(rows);
    if (type === 'leadCorrections') return updateLeadCorrections(body);
    if (type === 'updateLead') return updateLeadRow(body);
    if (type === 'saveKpiTarget') return saveKpiTarget(body, user);

    return json({ error: 'Invalid type' });
  } catch (err) {
    return json({ error: String(err) });
  }
}

function getLeadUploadContext(body) {
  const keys = body.rows || [];
  if (!keys.length) return json({ success: true, leads: [], firstRaw: [] });

  const phones = {};
  const consultingNumbers = {};
  const phoneDates = {};
  keys.forEach(function(item) {
    const phone = normalizePhone(item.phone || item._parsed_phone || '');
    const date = String(item.date || item._parsed_date || '').slice(0, 10);
    const consultingNumber = String(item.consultingNumber || item.consulting_number || '').trim();
    if (phone) phones[phone] = true;
    if (consultingNumber) consultingNumbers[consultingNumber] = true;
    if (phone && date) phoneDates[phone + '_' + date] = true;
  });

  const leads = readRowsAsObjects(SHEETS.leads).filter(function(row) {
    const phone = normalizePhone(row.phone || row._parsed_phone || row['연락처'] || row['휴대폰번호'] || '');
    const consultingNumber = String(row.consultingNumber || row.consulting_number || row['컨설팅번호'] || '').trim();
    return Boolean((phone && phones[phone]) || (consultingNumber && consultingNumbers[consultingNumber]));
  });

  const firstRaw = readRowsAsObjects(SHEETS.firstRaw).filter(function(row) {
    const phone = normalizePhone(row._parsed_phone || row.phone || row['연락처'] || row['휴대폰번호'] || '');
    const date = String(row._parsed_date || row.date || row['등록일시'] || row['등록 일시'] || row['접수일시'] || '').slice(0, 10);
    return Boolean(phone && date && phoneDates[phone + '_' + date]);
  });

  return json({ success: true, leads: leads, firstRaw: firstRaw });
}

function saveKpiTarget(body, user) {
  if (!requireMaster(user)) return json({ error: 'forbidden' });
  const target = body.target || {};
  const month = String(target.month || '').slice(0, 7);
  const minDaily = Math.max(0, Math.round(Number(target.minDaily) || 0));
  const stretchDaily = Math.max(minDaily, Math.round(Number(target.stretchDaily) || 0));
  const targetCpl = Math.max(0, Math.round(Number(target.targetCpl) || 0));
  if (!/^\d{4}-\d{2}$/.test(month)) return json({ error: 'invalid target month' });
  if (minDaily <= 0 || stretchDaily <= 0) return json({ error: 'invalid target value' });

  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const sheet = ensureKpiTargetSheet();
    const values = sheet.getDataRange().getDisplayValues();
    const headers = values[0] || [];
    let rowNumber = -1;
    for (let row = 1; row < values.length; row++) {
      if (String(values[row][0] || '').slice(0, 7) === month) {
        rowNumber = row + 1;
        break;
      }
    }
    const nextTarget = {
      month: month,
      minDaily: minDaily,
      stretchDaily: stretchDaily,
      targetCpl: targetCpl || '',
      updatedBy: String(user.name || user.id || ''),
      updatedAt: new Date().toISOString()
    };
    const output = [headers.map(function(header) {
      return nextTarget[header] === undefined ? '' : nextTarget[header];
    })];
    if (rowNumber > 0) sheet.getRange(rowNumber, 1, 1, headers.length).setValues(output);
    else sheet.getRange(sheet.getLastRow() + 1, 1, headers.length).setValues(output);
    return json({ success: true, target: {
      month: month,
      minDaily: minDaily,
      stretchDaily: stretchDaily,
      targetCpl: targetCpl,
      updatedBy: String(user.name || user.id || '')
    }});
  } finally {
    lock.releaseLock();
  }
}

function appendRowsUnique(sheetName, rows, type) {
  const book = SpreadsheetApp.getActive();
  let sheet = book.getSheetByName(sheetName);
  if (!sheet) sheet = book.insertSheet(sheetName);
  if (!rows || rows.length === 0) return json({ success: true, count: 0, skipped: 0 });

  let lastCol = sheet.getLastColumn();
  let headers = lastCol > 0 ? sheet.getRange(1, 1, 1, lastCol).getDisplayValues()[0].filter(String) : [];

  // 업로드 데이터에 새 컬럼이 있으면 시트 헤더를 자동 확장
  const incomingKeys = [];
  rows.forEach(row => Object.keys(row).forEach(k => { if (incomingKeys.indexOf(k) === -1) incomingKeys.push(k); }));
  incomingKeys.forEach(k => { if (headers.indexOf(k) === -1) headers.push(k); });

  if (headers.length === 0) return json({ error: 'No headers found' });
  sheet.getRange(1, 1, 1, headers.length).setValues([headers]);

  const existing = buildExistingKeySet(sheet, headers, type);
  const toAppend = [];
  const seenIncoming = {};

  rows.forEach(row => {
    const key = makeUniqueKey(row, type);
    if (!key) return;
    if (existing[key] || seenIncoming[key]) return;
    seenIncoming[key] = true;
    toAppend.push(row);
  });

  if (toAppend.length === 0) return json({ success: true, count: 0, skipped: rows.length });

  const values = toAppend.map(row => headers.map(h => row[h] === undefined || row[h] === null ? '' : row[h]));
  sheet.getRange(sheet.getLastRow() + 1, 1, values.length, headers.length).setValues(values);
  return json({ success: true, count: values.length, skipped: rows.length - values.length });
}

function upsertRowsUnique(sheetName, rows, type) {
  const book = SpreadsheetApp.getActive();
  let sheet = book.getSheetByName(sheetName);
  if (!sheet) sheet = book.insertSheet(sheetName);
  if (!rows || rows.length === 0) return json({ success: true, count: 0, updated: 0, skipped: 0 });

  let lastCol = sheet.getLastColumn();
  let headers = lastCol > 0 ? sheet.getRange(1, 1, 1, lastCol).getDisplayValues()[0].filter(String) : [];
  const incomingKeys = [];
  rows.forEach(row => Object.keys(row).forEach(k => { if (incomingKeys.indexOf(k) === -1) incomingKeys.push(k); }));
  incomingKeys.forEach(k => { if (headers.indexOf(k) === -1) headers.push(k); });
  if (headers.length === 0) return json({ error: 'No headers found' });
  sheet.getRange(1, 1, 1, headers.length).setValues([headers]);

  const values = sheet.getDataRange().getDisplayValues();
  const existingRows = {};
  for (let r = 1; r < values.length; r++) {
    const obj = {};
    headers.forEach((h, i) => obj[h] = values[r][i] || '');
    const key = makeUniqueKey(obj, type);
    if (key && existingRows[key] === undefined) existingRows[key] = r + 1;
  }

  let inserted = 0;
  let updated = 0;
  let unchanged = 0;
  const seenIncoming = {};
  const updatesByRow = {};
  const rowsToAppend = [];
  const ignoreUploadTimestamp = type === 'firstRaw' || type === 'secondRaw';
  rows.forEach(row => {
    const key = makeUniqueKey(row, type);
    if (!key || seenIncoming[key]) return;
    seenIncoming[key] = true;
    const output = headers.map(h => row[h] === undefined || row[h] === null ? '' : row[h]);
    if (existingRows[key]) {
      const rowNumber = existingRows[key];
      const previous = values[rowNumber - 1];
      const changed = headers.some(function(header, column) {
        if (ignoreUploadTimestamp && header === '_uploadedAt') return false;
        return String(previous[column] == null ? '' : previous[column]) !== String(output[column]);
      });
      if (!changed) {
        unchanged++;
        return;
      }
      updatesByRow[rowNumber] = output;
      updated++;
    } else {
      rowsToAppend.push(output);
      inserted++;
    }
  });

  // 기존 행은 연속된 구간끼리 묶어서 저장한다. 행마다 setValues를 호출하면
  // 대용량 컨설팅리스트에서 Google Sheets 왕복이 수백 번 발생한다.
  const updateRows = Object.keys(updatesByRow).map(Number).sort((a, b) => a - b);
  let groupStart = -1;
  let groupValues = [];
  let previousRow = -1;
  const flushUpdateGroup = function() {
    if (groupStart < 0 || !groupValues.length) return;
    sheet.getRange(groupStart, 1, groupValues.length, headers.length).setValues(groupValues);
    groupStart = -1;
    groupValues = [];
    previousRow = -1;
  };

  updateRows.forEach(rowNumber => {
    if (groupStart < 0 || rowNumber !== previousRow + 1) {
      flushUpdateGroup();
      groupStart = rowNumber;
    }
    groupValues.push(updatesByRow[rowNumber]);
    previousRow = rowNumber;
  });
  flushUpdateGroup();

  if (rowsToAppend.length) {
    sheet.getRange(sheet.getLastRow() + 1, 1, rowsToAppend.length, headers.length).setValues(rowsToAppend);
  }

  return json({ success: true, count: inserted, updated: updated, unchanged: unchanged, skipped: rows.length - inserted - updated });
}

function replaceAdSpendRows(rows) {
  const sheet = SpreadsheetApp.getActive().getSheetByName(SHEETS.adSpend);
  if (!sheet) return json({ error: 'Sheet not found: ' + SHEETS.adSpend });
  if (!rows || rows.length === 0) return json({ success: true, count: 0, deleted: 0 });

  let lastCol = sheet.getLastColumn();
  let headers = lastCol > 0 ? sheet.getRange(1, 1, 1, lastCol).getDisplayValues()[0].filter(String) : [];
  const incomingKeys = [];
  rows.forEach(row => Object.keys(row).forEach(k => { if (incomingKeys.indexOf(k) === -1) incomingKeys.push(k); }));
  incomingKeys.forEach(k => { if (headers.indexOf(k) === -1) headers.push(k); });
  if (headers.length === 0) return json({ error: 'No headers found' });
  sheet.getRange(1, 1, 1, headers.length).setValues([headers]);

  const dateIdx = headers.indexOf('date');
  const channelIdx = headers.indexOf('channel');
  if (dateIdx < 0 || channelIdx < 0) return json({ error: 'AD_SPEND date/channel column not found' });

  const targets = {};
  rows.forEach(row => {
    const date = String(row.date || '').slice(0, 10);
    const channel = String(row.channel || '').toLowerCase();
    if (date && channel) targets[date + '_' + channel] = true;
  });

  const values = sheet.getDataRange().getDisplayValues();
  let deleted = 0;
  for (let r = values.length - 1; r >= 1; r--) {
    const date = String(values[r][dateIdx] || '').slice(0, 10);
    const channel = String(values[r][channelIdx] || '').toLowerCase();
    if (targets[date + '_' + channel]) {
      sheet.deleteRow(r + 1);
      deleted++;
    }
  }

  const rowsToAppend = rows.filter(function(row) {
    return Number(String(row.amount || row.cost || row['광고비'] || row['비용'] || 0).replace(/[^0-9]/g, '')) > 0;
  });
  if (rowsToAppend.length === 0) {
    return json({ success: true, count: 0, skipped: rows.length, deleted: deleted, replaced: true });
  }

  const appendResult = JSON.parse(appendRowsUnique(SHEETS.adSpend, rowsToAppend, 'adSpend').getContent());
  appendResult.deleted = deleted;
  appendResult.replaced = true;
  return json(appendResult);
}

function updateAdSpendRow(body) {
  const sheet = SpreadsheetApp.getActive().getSheetByName(SHEETS.adSpend);
  if (!sheet) return json({ error: 'Sheet not found: ' + SHEETS.adSpend });

  const original = body.original || {};
  const patch = body.patch || {};
  const originalKey = adSpendRowKey(original);
  if (!originalKey) return json({ error: 'original ad spend key required' });

  let lastCol = sheet.getLastColumn();
  let headers = lastCol > 0 ? sheet.getRange(1, 1, 1, lastCol).getDisplayValues()[0].filter(String) : [];
  Object.keys(patch).forEach(k => { if (headers.indexOf(k) === -1) headers.push(k); });
  sheet.getRange(1, 1, 1, headers.length).setValues([headers]);

  const values = sheet.getDataRange().getDisplayValues();
  let targetRow = -1;
  for (let r = 1; r < values.length; r++) {
    const row = {};
    headers.forEach((h, i) => row[h] = values[r][i] || '');
    if (adSpendRowKey(row) === originalKey) {
      targetRow = r + 1;
      break;
    }
  }
  if (targetRow < 0) return json({ error: 'matching ad spend not found' });

  const next = {};
  headers.forEach((h, i) => next[h] = values[targetRow - 1][i] || '');
  Object.keys(patch).forEach(k => next[k] = patch[k]);
  const nextKey = adSpendRowKey(next);

  for (let r = 1; r < values.length; r++) {
    if (r + 1 === targetRow) continue;
    const row = {};
    headers.forEach((h, i) => row[h] = values[r][i] || '');
    if (adSpendRowKey(row) === nextKey) return json({ error: 'duplicate ad spend' });
  }

  Object.keys(patch).forEach(k => {
    sheet.getRange(targetRow, headers.indexOf(k) + 1).setValue(patch[k]);
  });
  return json({ success: true, updated: 1 });
}

function deleteAdSpendRow(body) {
  const sheet = SpreadsheetApp.getActive().getSheetByName(SHEETS.adSpend);
  if (!sheet) return json({ error: 'Sheet not found: ' + SHEETS.adSpend });

  const originalKey = adSpendRowKey(body.original || {});
  if (!originalKey) return json({ error: 'original ad spend key required' });
  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getDisplayValues()[0];
  const values = sheet.getDataRange().getDisplayValues();

  for (let r = 1; r < values.length; r++) {
    const row = {};
    headers.forEach((h, i) => row[h] = values[r][i] || '');
    if (adSpendRowKey(row) === originalKey) {
      sheet.deleteRow(r + 1);
      return json({ success: true, deleted: 1 });
    }
  }
  return json({ error: 'matching ad spend not found' });
}


function baseLeadStage(v) {

function dateTimeKey(v) {
  const text = String(v || '').replace('T', ' ').replace('Z', '').trim();
  const match = text.match(/^(\d{4}-\d{2}-\d{2})\s+(\d{1,2}):(\d{1,2})/);
  return match ? match[1] + ' ' + ('0' + match[2]).slice(-2) + ':' + ('0' + match[3]).slice(-2) : text.slice(0, 16);
}

function updateLeadRow(body) {
  const sheet = SpreadsheetApp.getActive().getSheetByName(SHEETS.leads);
  if (!sheet) return json({ error: 'Sheet not found: ' + SHEETS.leads });

  const phone = normalizePhone(body.phone || '');
  const stage = baseLeadStage(body.stage || '');
  const patch = body.patch || {};
  const targetDate = String(body.date || '').slice(0, 10);
  const targetRegisteredAt = dateTimeKey(body.registeredAt || '');
  if (!phone || !stage) return json({ error: 'phone/stage required' });

  let lastCol = sheet.getLastColumn();
  let headers = lastCol > 0 ? sheet.getRange(1, 1, 1, lastCol).getDisplayValues()[0].filter(String) : [];
  Object.keys(patch).forEach(k => { if (headers.indexOf(k) === -1) headers.push(k); });
  if (headers.indexOf('changeHistory') === -1) headers.push('changeHistory');
  sheet.getRange(1, 1, 1, headers.length).setValues([headers]);

  const values = sheet.getDataRange().getDisplayValues();
  if (!values || values.length <= 1) return json({ error: 'No leads data' });
  const h = values[0];
  const phoneIdx = h.indexOf('phone');
  const stageIdx = h.indexOf('stage') >= 0 ? h.indexOf('stage') : h.indexOf('dbTier');
  const dateIdx = h.indexOf('date');
  const registeredAtIdx = h.indexOf('registeredAt');
  if (phoneIdx < 0 || stageIdx < 0) return json({ error: 'phone/stage column not found' });

  let matchedRow = -1;
  let fallbackRow = -1;
  for (let r = 1; r < values.length; r++) {
    const rowPhone = normalizePhone(values[r][phoneIdx]);
    const rowStage = baseLeadStage(values[r][stageIdx]);
    const rowDate = dateIdx >= 0 ? String(values[r][dateIdx] || '').slice(0, 10) : '';
    if (rowPhone !== phone || rowStage !== stage || (targetDate && rowDate !== targetDate)) continue;
    fallbackRow = r;
    const rowRegisteredAt = registeredAtIdx >= 0 ? dateTimeKey(values[r][registeredAtIdx]) : '';
    if (!targetRegisteredAt || (rowRegisteredAt && rowRegisteredAt === targetRegisteredAt)) {
      matchedRow = r;
      break;
    }
  }
  if (matchedRow < 0) matchedRow = fallbackRow;
  if (matchedRow < 0) return json({ success: false, updated: 0, error: 'matching lead not found' });

  const now = new Date().toISOString();
  const operator = patch.operator || patch.updatedBy || '';
  const historyIdx = headers.indexOf('changeHistory');
  const oldHistory = historyIdx >= 0 ? String(values[matchedRow][historyIdx] || '') : '';
  const changedKeys = Object.keys(patch).filter(k => k !== 'updatedAt' && k !== 'updatedBy' && patch[k] !== undefined && patch[k] !== null).join(', ');
  const action = String(patch.status || '').toLowerCase() === 'deleted' ? '삭제' : '수정';
  const historyLine = '[' + now.replace('T',' ').slice(0,16) + '] ' + (operator || '작업자미상') + ' ' + action + ': ' + changedKeys;
  patch.updatedAt = now;
  patch.changeHistory = oldHistory ? oldHistory + '\n' + historyLine : historyLine;

  Object.keys(patch).forEach(k => {
    const col = h.indexOf(k) >= 0 ? h.indexOf(k) : headers.indexOf(k);
    sheet.getRange(matchedRow + 1, col + 1).setValue(patch[k]);
  });
  return json({ success: true, updated: 1 });
}

function updateLeadCorrections(body) {
  const sheet = SpreadsheetApp.getActive().getSheetByName(SHEETS.leads);
  if (!sheet) return json({ error: 'Sheet not found: ' + SHEETS.leads });

  const items = body.rows || [];
  if (!items.length) return json({ success: true, updated: 0, requested: 0 });

  let lastCol = sheet.getLastColumn();
  let headers = lastCol > 0 ? sheet.getRange(1, 1, 1, lastCol).getDisplayValues()[0].filter(String) : [];
  if (headers.indexOf('changeHistory') === -1) headers.push('changeHistory');
  items.forEach(item => {
    const patch = item.patch || {};
    Object.keys(patch).forEach(k => { if (headers.indexOf(k) === -1) headers.push(k); });
  });
  sheet.getRange(1, 1, 1, headers.length).setValues([headers]);

  const values = sheet.getDataRange().getDisplayValues();
  if (!values || values.length <= 1) return json({ error: 'No leads data' });
  values[0] = headers.slice();
  for (let r = 0; r < values.length; r++) {
    while (values[r].length < headers.length) values[r].push('');
  }

  const phoneIdx = headers.indexOf('phone');
  const stageIdx = headers.indexOf('stage') >= 0 ? headers.indexOf('stage') : headers.indexOf('dbTier');
  const dateIdx = headers.indexOf('date');
  const registeredAtIdx = headers.indexOf('registeredAt');
  const consultingIdx = headers.indexOf('consultingNumber');
  const historyIdx = headers.indexOf('changeHistory');
  if (phoneIdx < 0 || stageIdx < 0) return json({ error: 'phone/stage column not found' });

  let updated = 0;
  const changedRowNumbers = {};
  items.forEach(item => {
    const phone = normalizePhone(item.phone || '');
    const stage = baseLeadStage(item.stage || '');
    const targetDate = String(item.matchDate || item.date || '').slice(0, 10);
    const targetRegisteredAt = dateTimeKey(item.registeredAt || '');
    const consultingNumber = String(item.consultingNumber || '').trim();
    if (!phone || !stage) return;

    let matchedRow = -1;
    let fallbackRow = -1;
    for (let r = 1; r < values.length; r++) {
      if (consultingNumber && consultingIdx >= 0 && String(values[r][consultingIdx] || '').trim() === consultingNumber) {
        matchedRow = r;
        break;
      }
      const rowPhone = normalizePhone(values[r][phoneIdx]);
      const rowStage = baseLeadStage(values[r][stageIdx]);
      const rowDate = dateIdx >= 0 ? String(values[r][dateIdx] || '').slice(0, 10) : '';
      if (rowPhone !== phone || rowStage !== stage || (targetDate && rowDate !== targetDate)) continue;
      fallbackRow = r;
      const rowRegisteredAt = registeredAtIdx >= 0 ? dateTimeKey(values[r][registeredAtIdx]) : '';
      if (!targetRegisteredAt || !rowRegisteredAt || rowRegisteredAt === targetRegisteredAt) {
        matchedRow = r;
        break;
      }
    }
    if (matchedRow < 0) matchedRow = fallbackRow;
    if (matchedRow < 0) return;

    const patch = item.patch || {};
    const now = new Date().toISOString();
    const operator = patch.operator || patch.updatedBy || '';
    const changedKeys = Object.keys(patch).filter(k => k !== 'updatedAt' && k !== 'updatedBy' && patch[k] !== undefined && patch[k] !== null).join(', ');
    const oldHistory = historyIdx >= 0 ? String(values[matchedRow][historyIdx] || '') : '';
    const historyLine = '[' + now.replace('T', ' ').slice(0, 16) + '] ' + (operator || 'system') + ' update: ' + changedKeys;
    patch.updatedAt = now;
    patch.changeHistory = oldHistory ? oldHistory + '\n' + historyLine : historyLine;

    Object.keys(patch).forEach(k => {
      const col = headers.indexOf(k);
      if (col >= 0) values[matchedRow][col] = patch[k] == null ? '' : patch[k];
    });
    changedRowNumbers[matchedRow] = true;
    updated++;
  });

  const changedRows = Object.keys(changedRowNumbers).map(Number).sort(function(a, b) { return a - b; });
  let start = -1;
  let group = [];
  let last = -1;
  const flush = function() {
    if (start < 0) return;
    sheet.getRange(start + 1, 1, group.length, headers.length).setValues(group);
    start = -1;
    group = [];
  };
  changedRows.forEach(function(rowNumber) {
    if (start < 0 || rowNumber !== last + 1) {
      flush();
      start = rowNumber;
    }
    group.push(values[rowNumber]);
    last = rowNumber;
  });
  flush();
  return json({ success: true, updated: updated, requested: items.length });
}

function buildExistingKeySet(sheet, headers, type) {
  const set = {};
  const values = sheet.getDataRange().getDisplayValues();
  if (!values || values.length <= 1) return set;
  values.shift();
  values.forEach(row => {
    const obj = {};
    headers.forEach((h, i) => obj[h] = row[i]);
    const key = makeUniqueKey(obj, type);
    if (key) set[key] = true;
  });
  return set;
}

function normalizePhone(v) {
  let phone = String(v || '').replace(/[^0-9]/g, '');
  if (phone.length === 10 && phone.indexOf('10') === 0) phone = '0' + phone;
  return phone;
}

function makeUniqueKey(row, type) {
  if (type === 'leads') {
    const phone = normalizePhone(row.phone || row._parsed_phone || row['연락처'] || row['휴대폰번호'] || row['휴대폰 번호']);
    const stage = String(row.stage || row.dbTier || row._parsed_stage || '').toLowerCase();
    const date = String(row.date || row._parsed_date || row['등록일시'] || row['등록 일시'] || row['접수일시'] || '').slice(0, 10);
    return phone && stage && date ? phone + '_' + stage + '_' + date : (phone && stage ? phone + '_' + stage : '');
  }

  if (type === 'firstRaw' || type === 'secondRaw') {
    const phone = normalizePhone(row._parsed_phone || row.phone || row['연락처'] || row['휴대폰번호'] || row['휴대폰 번호']);
    const stage = String(row._parsed_stage || row.stage || row.dbTier || (type === 'secondRaw' ? 'second' : 'first')).toLowerCase();
    const date = String(row._parsed_date || row.date || row['등록일시'] || row['등록 일시'] || row['접수일시'] || '').slice(0, 10);
    return phone && stage ? type + '_' + phone + '_' + stage + '_' + date : '';
  }

  if (type === 'adSpend') {
    const date = String(row.date || row['날짜'] || '').slice(0, 10);
    const channel = String(row.channel || row['매체'] || '').toLowerCase();
    const subChannel = String(row.subChannel || row['상세매체'] || '').toLowerCase();
    const campaign = String(row.campaign || row['캠페인'] || '').toLowerCase();
    return date && channel ? date + '_' + channel + '_' + subChannel + '_' + campaign : '';
  }

  if (type === 'projects') {
    const projectNumber = String(row.projectNumber || row['프로젝트번호'] || '').trim().toLowerCase();
    const consultingNumber = String(row.consultingNumber || row['컨설팅번호'] || '').trim().toLowerCase();
    const phone = normalizePhone(row.phone || row['연락처'] || row['전화번호']);
    if (projectNumber) return 'project_' + projectNumber;
    if (consultingNumber) return 'consult_' + consultingNumber;
    return phone ? 'phone_' + phone : '';
  }

  return '';
}

const USER_HEADERS = ['id', 'name', 'passwordHash', 'salt', 'role', 'permissions', 'active', 'createdAt', 'updatedAt'];
const SESSION_CACHE_TTL_SECONDS = 21600;
const SESSION_TTL_MS = 30 * 24 * 60 * 60 * 1000;
const SESSION_PROPERTY_PREFIX = 'admin_session_v2_';
const SESSION_TOKEN_VERSION = 3;
const SALES_SUPPORT_MENUS = ['/db-manage', '/upload-db', '/region'];
const VALID_MENUS = [
  '/dashboard', '/channels', '/kpi', '/funnel', '/region', '/db-manage',
  '/upload-db', '/upload-spend', '/manage-spend', '/upload-projects', '/sales-performance', '/admin-users'
];

function ensureUsersSheet() {
  const book = SpreadsheetApp.getActive();
  let sheet = book.getSheetByName(SHEETS.users);
  if (!sheet) sheet = book.insertSheet(SHEETS.users);
  if (sheet.getLastRow() === 0) {
    sheet.getRange(1, 1, 1, USER_HEADERS.length).setValues([USER_HEADERS]);
    sheet.setFrozenRows(1);
  }
  return sheet;
}

function getUsers() {
  const sheet = ensureUsersSheet();
  if (sheet.getLastRow() <= 1) return [];
  const values = sheet.getRange(2, 1, sheet.getLastRow() - 1, USER_HEADERS.length).getDisplayValues();
  return values.filter(row => String(row[0] || '').trim()).map((row, index) => {
    const user = { rowNumber: index + 2 };
    USER_HEADERS.forEach((header, column) => user[header] = row[column]);
    user.active = String(user.active).toLowerCase() !== 'false';
    user.permissions = parsePermissions(user.permissions);
    return user;
  });
}

function findUser(id) {
  const key = String(id || '').trim().toLowerCase();
  if (!key) return null;
  const users = getUsers();
  for (let i = 0; i < users.length; i++) {
    if (String(users[i].id || '').trim().toLowerCase() === key) return users[i];
  }
  return null;
}

function parsePermissions(value) {
  if (Array.isArray(value)) return value.map(String).filter(Boolean);
  return String(value || '').split(/[|,]/).map(value => value.trim()).filter(Boolean);
}

function normalizedPermissions(role, permissions) {
  if (role === 'master') return ['*'];
  if (role === 'sales_support') return SALES_SUPPORT_MENUS.slice();
  const requested = parsePermissions(permissions);
  return VALID_MENUS.filter(menu => requested.indexOf(menu) >= 0);
}

function publicUser(user) {
  return {
    id: String(user.id || ''),
    name: String(user.name || ''),
    role: String(user.role || 'custom'),
    permissions: parsePermissions(user.permissions),
    active: user.active !== false && String(user.active).toLowerCase() !== 'false',
    createdAt: String(user.createdAt || ''),
    updatedAt: String(user.updatedAt || '')
  };
}

function hashPassword(password, salt) {
  const bytes = Utilities.computeDigest(
    Utilities.DigestAlgorithm.SHA_256,
    String(salt) + ':' + String(password),
    Utilities.Charset.UTF_8
  );
  return Utilities.base64EncodeWebSafe(bytes);
}

function createSession(userId) {
  const user = findUser(userId);
  if (!user) throw new Error('user not found');
  const payloadPart = sessionBase64Encode(JSON.stringify({
    version: SESSION_TOKEN_VERSION,
    userId: String(user.id),
    expiresAt: Date.now() + SESSION_TTL_MS,
    nonce: Utilities.getUuid()
  }));
  return payloadPart + '.' + signSessionPayload(payloadPart, user);
}

function removeSession(token) {
  if (!token) return;
  const key = String(token);
  CacheService.getScriptCache().remove('admin_session_' + key);
  PropertiesService.getScriptProperties().deleteProperty(SESSION_PROPERTY_PREFIX + key);
}

function readSession(token) {
  if (!token) return null;
  const key = String(token);
  const parts = key.split('.');
  if (parts.length === 2) {
    try {
      const payload = JSON.parse(sessionBase64Decode(parts[0]));
      if (Number(payload.version || 0) !== SESSION_TOKEN_VERSION) return null;
      if (!payload.userId || Number(payload.expiresAt || 0) <= Date.now()) return null;
      const user = findUser(payload.userId);
      if (!user || !publicUser(user).active) return null;
      const expected = signSessionPayload(parts[0], user);
      if (!sessionSafeEquals(parts[1], expected)) return null;
      payload.user = user;
      return payload;
    } catch (error) {
      return null;
    }
  }

  // 이전 배포에서 발급한 저장형 토큰은 만료 전까지 호환합니다.
  const cacheKey = 'admin_session_' + key;
  const propertyKey = SESSION_PROPERTY_PREFIX + key;
  let raw = CacheService.getScriptCache().get(cacheKey);
  if (!raw) raw = PropertiesService.getScriptProperties().getProperty(propertyKey);
  if (!raw) return null;

  let session = null;
  try {
    session = JSON.parse(raw);
  } catch (error) {
    // 이전 캐시 방식의 세션은 배포 직후 한 번만 호환합니다.
    session = { userId: String(raw), expiresAt: Date.now() + SESSION_TTL_MS };
  }

  if (!session.userId || Number(session.expiresAt || 0) <= Date.now()) {
    removeSession(key);
    return null;
  }

  PropertiesService.getScriptProperties().setProperty(propertyKey, JSON.stringify(session));
  CacheService.getScriptCache().put(cacheKey, JSON.stringify(session), SESSION_CACHE_TTL_SECONDS);
  return session;
}

function sessionUser(token) {
  const session = readSession(token);
  if (!session) return null;
  const user = session.user || findUser(session.userId);
  if (!user || !publicUser(user).active) {
    removeSession(token);
    return null;
  }
  return user;
}

function sessionBase64Encode(value) {
  return Utilities.base64EncodeWebSafe(String(value), Utilities.Charset.UTF_8).replace(/=+$/g, '');
}

function sessionBase64Decode(value) {
  let encoded = String(value || '');
  while (encoded.length % 4) encoded += '=';
  return Utilities.newBlob(Utilities.base64DecodeWebSafe(encoded)).getDataAsString('UTF-8');
}

function sessionSigningKey(user) {
  const spreadsheetId = SpreadsheetApp.getActive().getId();
  return spreadsheetId + ':' + String(user.id || '') + ':' + String(user.passwordHash || '') + ':' + String(user.salt || '');
}

function signSessionPayload(payloadPart, user) {
  const signature = Utilities.computeHmacSha256Signature(
    String(payloadPart),
    sessionSigningKey(user),
    Utilities.Charset.UTF_8
  );
  return Utilities.base64EncodeWebSafe(signature).replace(/=+$/g, '');
}

function sessionSafeEquals(left, right) {
  const a = String(left || '');
  const b = String(right || '');
  if (a.length !== b.length) return false;
  let mismatch = 0;
  for (let index = 0; index < a.length; index++) {
    mismatch |= a.charCodeAt(index) ^ b.charCodeAt(index);
  }
  return mismatch === 0;
}

function authStatus(token) {
  const users = getUsers();
  if (users.length === 0) return json({ setupRequired: true, authVersion: SESSION_TOKEN_VERSION });
  const user = sessionUser(token);
  return json({ setupRequired: false, user: user ? publicUser(user) : null, authVersion: SESSION_TOKEN_VERSION });
}

function authSetup(body) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    if (getUsers().length > 0) return json({ error: 'setup already completed' });
    const id = String(body.id || '').trim();
    const name = String(body.name || '').trim() || id;
    const password = String(body.password || '');
    if (!id) return json({ error: 'id required' });
    if (password.length < 8) return json({ error: 'password must be at least 8 characters' });

    const now = new Date().toISOString();
    const salt = Utilities.getUuid();
    const user = {
      id: id,
      name: name,
      passwordHash: hashPassword(password, salt),
      salt: salt,
      role: 'master',
      permissions: ['*'],
      active: true,
      createdAt: now,
      updatedAt: now
    };
    appendUser(user);
    return json({ success: true, token: createSession(id), user: publicUser(user) });
  } finally {
    lock.releaseLock();
  }
}

function authLogin(body) {
  const user = findUser(body.id);
  if (!user || !publicUser(user).active) return json({ error: 'invalid credentials' });
  const actual = hashPassword(String(body.password || ''), user.salt);
  if (actual !== String(user.passwordHash || '')) return json({ error: 'invalid credentials' });
  return json({ success: true, token: createSession(user.id), user: publicUser(user), authVersion: SESSION_TOKEN_VERSION });
}

function authLogout(token) {
  removeSession(token);
  return json({ success: true });
}

function appendUser(user) {
  const sheet = ensureUsersSheet();
  const row = USER_HEADERS.map(header => {
    if (header === 'permissions') return parsePermissions(user[header]).join('|');
    return user[header] === undefined || user[header] === null ? '' : user[header];
  });
  sheet.getRange(sheet.getLastRow() + 1, 1, 1, USER_HEADERS.length).setValues([row]);
}

function requireMaster(user) {
  return user && String(user.role) === 'master' && publicUser(user).active;
}

function authListUsers(actor) {
  if (!requireMaster(actor)) return json({ error: 'forbidden' });
  return json({ users: getUsers().map(publicUser) });
}

function authCreateUser(actor, body) {
  if (!requireMaster(actor)) return json({ error: 'forbidden' });
  const id = String(body.id || '').trim();
  const name = String(body.name || '').trim() || id;
  const password = String(body.password || '');
  const role = ['master', 'sales_support', 'custom'].indexOf(String(body.role)) >= 0 ? String(body.role) : 'custom';
  if (!id) return json({ error: 'id required' });
  if (findUser(id)) return json({ error: 'id already exists' });
  if (password.length < 8) return json({ error: 'password must be at least 8 characters' });

  const now = new Date().toISOString();
  const salt = Utilities.getUuid();
  const user = {
    id: id,
    name: name,
    passwordHash: hashPassword(password, salt),
    salt: salt,
    role: role,
    permissions: normalizedPermissions(role, body.permissions),
    active: true,
    createdAt: now,
    updatedAt: now
  };
  appendUser(user);
  return json({ success: true, user: publicUser(user) });
}

function authUpdateUser(actor, body) {
  if (!requireMaster(actor)) return json({ error: 'forbidden' });
  const user = findUser(body.id);
  if (!user) return json({ error: 'user not found' });

  const isSelf = String(actor.id).toLowerCase() === String(user.id).toLowerCase();
  const role = body.role === undefined ? String(user.role) :
    (['master', 'sales_support', 'custom'].indexOf(String(body.role)) >= 0 ? String(body.role) : 'custom');
  const active = body.active === undefined ? publicUser(user).active : Boolean(body.active);
  if (isSelf && (role !== 'master' || !active)) return json({ error: 'cannot remove your own master access' });

  user.name = body.name === undefined ? user.name : (String(body.name || '').trim() || user.id);
  user.role = role;
  user.permissions = normalizedPermissions(role, body.permissions === undefined ? user.permissions : body.permissions);
  user.active = active;
  user.updatedAt = new Date().toISOString();
  if (String(body.password || '')) {
    if (String(body.password).length < 8) return json({ error: 'password must be at least 8 characters' });
    user.salt = Utilities.getUuid();
    user.passwordHash = hashPassword(String(body.password), user.salt);
  }

  const sheet = ensureUsersSheet();
  const row = USER_HEADERS.map(header => {
    if (header === 'permissions') return parsePermissions(user[header]).join('|');
    return user[header] === undefined || user[header] === null ? '' : user[header];
  });
  sheet.getRange(user.rowNumber, 1, 1, USER_HEADERS.length).setValues([row]);
  return json({ success: true, user: publicUser(user) });
}

function canAccessMenu(user, menu) {
  if (!user || VALID_MENUS.indexOf(menu) < 0) return false;
  if (String(user.role) === 'master') return true;
  const permissions = parsePermissions(user.permissions);
  return permissions.indexOf('*') >= 0 || permissions.indexOf(menu) >= 0;
}
